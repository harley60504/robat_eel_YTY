import time
import threading
import tkinter as tk
from tkinter import ttk
import numpy as np
import mujoco
import mujoco.viewer
from eel_env import EelEnv

class ControlPanel:
    def __init__(self):
        self.root = tk.Tk()
        self.root.title("Eel Control Panel - Stable Version")

        self.lock = threading.Lock()

        self.DEFAULTS = {
            "amp": 0.5,
            "freq": 1.0,
            "turn_bias": 0.0,
            "phase_step": 0.8
        }

        self.paused = True
        self.reset_requested = False
        self.current_speed = 0.0
        self.vars = {}
        self.is_alive = True  # 用於判斷程式是否正在關閉

        for attr, val in self.DEFAULTS.items():
            setattr(self, attr, val)

        frm = ttk.Frame(self.root, padding=15)
        frm.grid(row=0, column=0, sticky="nsew")

        self.status_var = tk.StringVar(value="PAUSED")
        self.speed_var = tk.StringVar(value="Speed: 0.00 m/s")
        
        ttk.Label(frm, textvariable=self.status_var, font=("Segoe UI", 12, "bold")).grid(row=0, column=0, pady=(0, 5), sticky="w")
        ttk.Label(frm, textvariable=self.speed_var, font=("Consolas", 12), foreground="blue").grid(row=1, column=0, pady=(0, 10), sticky="w")

        btn_frm = ttk.Frame(frm)
        btn_frm.grid(row=2, column=0, columnspan=3, sticky="ew", pady=5)
        
        ttk.Button(btn_frm, text="Run / Pause", command=self.toggle_pause).pack(side="left", expand=True, fill="x")
        ttk.Button(btn_frm, text="Reset All", command=self.request_reset).pack(side="left", expand=True, fill="x")
        ttk.Button(btn_frm, text="Quit", command=self.quit).pack(side="left", expand=True, fill="x")

        self._add_entry(frm, "擺動振幅 (Amplitude)", 3, "amp")
        self._add_entry(frm, "游泳頻率 (Frequency)", 4, "freq")
        self._add_entry(frm, "每節偏移 (Turn Bias)", 5, "turn_bias")
        self._add_entry(frm, "相位差 (Phase Step)", 6, "phase_step")

        self._update_ui_loop()
        self.root.protocol("WM_DELETE_WINDOW", self.quit)

    def _add_entry(self, parent, label_text, row, attr):
        ttk.Label(parent, text=label_text).grid(row=row, column=0, sticky="w", pady=5)
        var = tk.StringVar(value=str(getattr(self, attr)))
        self.vars[attr] = var
        
        def on_type(*args):
            try:
                val = float(var.get())
                with self.lock:
                    setattr(self, attr, val)
            except ValueError: pass

        var.trace_add("write", on_type)
        entry = ttk.Entry(parent, textvariable=var, width=10)
        entry.grid(row=row, column=1, sticky="w", pady=5, padx=10)

    def _update_ui_loop(self):
        if not self.is_alive: return
        try:
            with self.lock:
                self.speed_var.set(f"Speed: {self.current_speed:.3f} m/s")
            self.root.after(100, self._update_ui_loop)
        except: pass

    def toggle_pause(self):
        with self.lock:
            self.paused = not self.paused
            self.status_var.set("RUNNING" if not self.paused else "PAUSED")

    def request_reset(self):
        # 為了防止崩潰，我們確保在 Lock 保護下更新狀態
        with self.lock:
            self.reset_requested = True
            self.paused = True 
            self.current_speed = 0.0
            
            # 使用 root.after 確保 UI 更新是在 Tk 主執行緒執行
            self.root.after(0, self._reset_ui_vars)

    def _reset_ui_vars(self):
        """專門負責將 UI 元件恢復預設值"""
        if not self.is_alive: return
        try:
            for attr, value in self.DEFAULTS.items():
                self.vars[attr].set(str(value))
                with self.lock:
                    setattr(self, attr, value)
            self.status_var.set("PAUSED (Reset)")
        except: pass

    def quit(self):
        with self.lock:
            self.is_alive = False
        try:
            self.root.quit()
            self.root.destroy()
        except: pass

def run_mujoco(panel: ControlPanel, xml_path="eel.xml"):
    env = EelEnv(xml_path)
    env.reset()
    
    # 備份初始狀態
    initial_qpos = np.copy(env.data.qpos)
    last_pos = np.copy(env.data.qpos[0:2])
    last_time = env.data.time

    with mujoco.viewer.launch_passive(env.model, env.data) as viewer:
        viewer.cam.azimuth = 0
        viewer.cam.elevation = -90.0
        viewer.cam.distance = 13.0
        viewer.cam.lookat = [0, 0, 0]

        while viewer.is_running():
            # 檢查面板是否還活著
            with panel.lock:
                if not panel.is_alive: break
                paused = panel.paused
                amp = panel.amp
                freq = panel.freq
                turn_bias = panel.turn_bias
                phase_step = panel.phase_step
                do_reset = panel.reset_requested
                if do_reset:
                    panel.reset_requested = False

            # --- 重置邏輯：在 viewer.lock 中執行以確保線程安全 ---
            if do_reset:
                with viewer.lock():
                    mujoco.mj_resetData(env.model, env.data)
                    env.data.qpos[:] = initial_qpos
                    env.data.qvel[:] = 0
                    env.data.ctrl[:] = turn_bias
                    mujoco.mj_forward(env.model, env.data)
                    last_pos = np.copy(env.data.qpos[0:2])
                    last_time = env.data.time
                continue # 重置完跳過這一幀

            if not paused:
                # 物理與控制計算
                t = env.data.time
                num_joints = len(env.data.ctrl)
                ctrl = np.zeros(num_joints)
                for i in range(num_joints):
                    phase = -i * phase_step
                    envelope = (0.4 + 0.6 * (i / (num_joints - 1)))
                    oscillation = (amp * envelope) * np.sin(2 * np.pi * freq * t + phase)
                    ctrl[i] = turn_bias + oscillation
                
                with viewer.lock():
                    env.data.ctrl[:] = np.clip(ctrl, -1.2, 1.2)
                    mujoco.mj_step(env.model, env.data)

                # 速度計算
                curr_time = env.data.time
                dt = curr_time - last_time
                if dt >= 0.05:
                    curr_pos = env.data.qpos[0:2]
                    dist = np.linalg.norm(curr_pos - last_pos)
                    speed = dist / dt
                    with panel.lock:
                        panel.current_speed = speed
                    last_pos = np.copy(curr_pos)
                    last_time = curr_time
            else:
                with viewer.lock():
                    env.data.ctrl[:] = turn_bias
                    mujoco.mj_forward(env.model, env.data)
                with panel.lock:
                    panel.current_speed = 0.0

            viewer.sync()
            time.sleep(env.model.opt.timestep)

if __name__ == "__main__":
    panel = ControlPanel()
    sim_thread = threading.Thread(target=run_mujoco, args=(panel, "eel.xml"), daemon=True)
    sim_thread.start()
    panel.root.mainloop()
